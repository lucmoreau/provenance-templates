<script>
function showTab(tabId) {
  // Find the tab to show and its top-level container div
  const activeTab = document.getElementById(tabId);
  const container = activeTab.parentElement;

  // Hide only the tab contents within this container
  container.querySelectorAll('.tab-content').forEach(tab => {
    tab.classList.remove('active');
    tab.style.display = 'none';
  });

  // Remove active class from buttons within this container only
  container.querySelectorAll('.tab-button').forEach(button => {
    button.classList.remove('active');
    button.style.backgroundColor = 'transparent';
    button.style.color = '#666';
    button.style.borderBottom = '3px solid transparent';
  });

  // Show selected tab content
  activeTab.classList.add('active');
  activeTab.style.display = 'block';

  // Mark the clicked button as active
  const activeButton = event.target;
  activeButton.classList.add('active');
  activeButton.style.backgroundColor = 'white';
  activeButton.style.color = '#2c3e50';
  activeButton.style.borderBottom = '3px solid #3498db';
}

// Initialize first tab as active when page loads
document.onreadystatechange = function() {
  if (document.readyState === "complete") {
    console.log("Page is loaded completely!");
    document.addEventListener('DOMContentLoaded', function() {
  showTab('tab1');
});
  }
};



</script>
